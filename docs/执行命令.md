```bash
python3 - <<'PY'
import json, sqlite3, pathlib, shutil, subprocess, time
from collections import defaultdict
from datetime import datetime, timezone, timedelta

TZ = timezone(timedelta(hours=8))
NOW = time.time()
SINCE = int((NOW - 30 * 86400) * 1000)
DB = pathlib.Path("config/agents/itservicedeskagent/agent/openclaw-agent.sqlite")

def kind_peer(key):
    parts = (key or "").split(":")
    for i, part in enumerate(parts):
        if part in ("dm", "direct", "group") and i + 1 < len(parts):
            return ("dm" if part == "direct" else part), ":".join(parts[i + 1:])
    return "other", None

def decompress(blob):
    blob = bytes(blob)
    if blob[:4] != b"\x28\xb5\x2f\xfd":
        return blob
    try:
        import zstandard
        return zstandard.ZstdDecompressor().decompress(blob)
    except Exception:
        pass
    zstd = shutil.which("zstd")
    if not zstd:
        raise RuntimeError("zstd-missing")
    return subprocess.check_output([zstd, "-dc"], input=blob)

def events_of(text):
    out = []
    for line in text.splitlines():
        line = line.strip()
        if not line.startswith("{"):
            continue
        try:
            o = json.loads(line)
        except Exception:
            continue
        if not isinstance(o, dict):
            continue
        if isinstance(o.get("message"), dict):
            out.append(o)
            continue
        row = o.get("row") if isinstance(o.get("row"), dict) else o
        raw = row.get("event_json") if isinstance(row, dict) else None
        if isinstance(raw, str):
            try:
                out.append(json.loads(raw))
            except Exception:
                pass
        elif isinstance(raw, dict):
            out.append(raw)
    return out

def content_texts(content):
    if isinstance(content, str):
        return [content]
    if isinstance(content, list):
        found = []
        for item in content:
            if isinstance(item, str):
                found.append(item)
            elif isinstance(item, dict) and isinstance(item.get("text"), str):
                found.append(item["text"])
        return found
    return []

def oa_receipt(msg):
    if not isinstance(msg, dict):
        return None
    if (msg.get("toolName") or msg.get("name")) != "oa_workflow":
        return None
    for text in content_texts(msg.get("content")):
        try:
            body = json.loads(text)
        except Exception:
            continue
        if isinstance(body, dict) and isinstance(body.get("code"), str):
            rid = body.get("requestId") or body.get("requestid")
            return body["code"], rid if isinstance(rid, str) and rid else None
    return None

def day_of(ms):
    return datetime.fromtimestamp(ms / 1000, TZ).date().isoformat()

def bucket(n):
    if n <= 1:
        return "1"
    if n <= 5:
        return "2-5"
    if n <= 20:
        return "6-20"
    return "21+"

def group_sender(msg):
    oc = msg.get("__openclaw")
    if not isinstance(oc, dict):
        return None
    sender = oc.get("senderId")
    if isinstance(sender, str) and sender:
        return sender
    ident = oc.get("senderIdentity")
    if isinstance(ident, dict) and isinstance(ident.get("id"), str) and ident.get("id"):
        return ident["id"]
    return None

con = sqlite3.connect(f"file:{DB.resolve()}?mode=ro", uri=True)
print("report_date", datetime.now(TZ).date().isoformat())
print("window", day_of(SINCE), day_of(int(NOW * 1000)))

live_peers, live_peers_30 = set(), set()
per_peer = defaultdict(int)
daily = defaultdict(lambda: [0, set()])
counts = defaultdict(lambda: [0, 0, 0])
oa_codes = defaultdict(int)
submitted_ids, submitted_peers = set(), set()
group_senders = set()

for created, raw, kind, peer in con.execute("""
  SELECT e.created_at, e.event_json, c.kind, c.peer_id
  FROM transcript_events e
  JOIN session_windows w ON w.session_id = e.session_id
  LEFT JOIN conversations c ON c.conversation_id = w.primary_conversation_id
"""):
    try:
        ev = json.loads(raw)
    except Exception:
        continue
    msg = ev.get("message") if isinstance(ev, dict) else None
    if not isinstance(msg, dict):
        continue
    kind = kind or "unlinked"
    role = msg.get("role")
    injected = "runtimeContext" in msg
    slot = counts[kind]
    if role == "assistant":
        slot[1] += 1
    elif role == "user" and injected:
        slot[2] += 1
    elif role == "user":
        slot[0] += 1
        if kind == "direct" and peer:
            live_peers.add(peer)
            per_peer[peer] += 1
            if created and created >= SINCE:
                live_peers_30.add(peer)
                daily[day_of(created)][0] += 1
                daily[day_of(created)][1].add(peer)
        if kind == "group":
            sender = group_sender(msg)
            if sender:
                group_senders.add(sender)
    receipt = oa_receipt(msg)
    if receipt:
        code, rid = receipt
        oa_codes[code] += 1
        if code == "SUBMITTED":
            if rid:
                submitted_ids.add(rid)
            if kind == "direct" and peer:
                submitted_peers.add(peer)

usage = con.execute("""
  SELECT COUNT(*),
         COALESCE(SUM(json_extract(event_json,'$.message.usage.input')), 0),
         COALESCE(SUM(json_extract(event_json,'$.message.usage.output')), 0),
         SUM(CASE WHEN json_extract(event_json,'$.message.usage.input') IS NOT NULL
                    OR json_extract(event_json,'$.message.usage.totalTokens') IS NOT NULL
                  THEN 1 ELSE 0 END)
  FROM transcript_events
""").fetchone()

archive_sessions = con.execute("SELECT COUNT(*) FROM session_transcript_archives").fetchone()[0]
arch_user = arch_asst = arch_err = 0
arch_peers, arch_peers_30 = set(), set()
arch_oa = defaultdict(int)
arch_submitted_ids = set()
for session_key, blob in con.execute(
        "SELECT session_key, archive_blob FROM session_transcript_archives"):
    kind, peer = kind_peer(session_key)
    try:
        text = decompress(blob).decode("utf-8")
    except Exception as e:
        arch_err += 1
        print(f"archive_error {type(e).__name__}")
        continue
    for ev in events_of(text):
        msg = ev.get("message") if isinstance(ev, dict) else None
        if not isinstance(msg, dict):
            continue
        role = msg.get("role")
        ts = msg.get("timestamp") or ev.get("timestamp")
        if role == "user" and "runtimeContext" not in msg:
            arch_user += 1
            if kind == "dm" and peer:
                arch_peers.add(peer)
                if isinstance(ts, (int, float)) and ts >= SINCE:
                    arch_peers_30.add(peer)
        elif role == "assistant":
            arch_asst += 1
        receipt = oa_receipt(msg)
        if not receipt:
            continue
        arch_oa[receipt[0]] += 1
        if receipt[0] == "SUBMITTED" and receipt[1]:
            arch_submitted_ids.add(receipt[1])

hist = defaultdict(int)
for n in per_peer.values():
    hist[bucket(n)] += 1

print(f"people_live={len(live_peers)}")
print(f"people_live_30d={len(live_peers_30)}")
print(f"people_archive_only={len(arch_peers - live_peers)}")
print(f"people_archive_only_30d={len((arch_peers - live_peers) & arch_peers_30)}")
print(f"people_total={len(live_peers | arch_peers)}")
print(f"people_submitted_live={len(submitted_peers)}")
print(f"people_live_without_submit={len(live_peers - submitted_peers)}")
for kind in ("direct", "group"):
    rows_n, distinct_n = con.execute(
        "SELECT COUNT(*), COUNT(DISTINCT peer_id) FROM conversations WHERE kind=?",
        (kind,),
    ).fetchone()
    print(f"conversations_{kind}={rows_n} distinct={distinct_n}")
print(f"group_senders={len(group_senders)}")
print(f"group_senders_not_in_live_peers={len(group_senders - live_peers)}")
print(
    "user_live "
    f"direct={counts['direct'][0]} group={counts['group'][0]} "
    f"unlinked={counts['unlinked'][0]} injected={sum(v[2] for v in counts.values())}"
)
print(
    "assistant_live "
    f"direct={counts['direct'][1]} group={counts['group'][1]} unlinked={counts['unlinked'][1]}"
)
print(f"archive_sessions={archive_sessions}")
print(f"archive_user_messages={arch_user} archive_assistant_messages={arch_asst} archive_errors={arch_err}")
print(f"user_direct_total={counts['direct'][0] + arch_user}")
print(f"assistant_direct_total={counts['direct'][1] + arch_asst}")
print(f"user_live_30d={sum(v[0] for v in daily.values())}")
print(f"busiest_day_people={max((len(v[1]) for v in daily.values()), default=0)}")
print("messages_per_live_person " + " ".join(f"{k}={hist[k]}" for k in ("1", "2-5", "6-20", "21+")))
print("daily_live_direct")
for day in sorted(daily):
    n, peers = daily[day]
    print(f"  {day} messages={n} people={len(peers)}")
print("oa_live " + " ".join(f"{k}={v}" for k, v in sorted(oa_codes.items())))
print(f"oa_live_receipts={sum(oa_codes.values())}")
print(f"oa_live_submitted_request_ids={len(submitted_ids)}")
print("oa_archive " + (" ".join(f"{k}={v}" for k, v in sorted(arch_oa.items())) or "none"))
print(f"oa_archive_submitted_request_ids={len(arch_submitted_ids)}")
print(f"oa_submitted_request_ids_total={len(submitted_ids | arch_submitted_ids)}")
print(f"tokens_events={usage[0]} input={usage[1]} output={usage[2]} events_with_usage={usage[3]}")
```
